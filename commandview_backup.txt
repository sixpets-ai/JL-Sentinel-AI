// Preload voices on first import.
if (typeof window !== "undefined" && window.speechSynthesis) {
  window.speechSynthesis.getVoices();
  window.speechSynthesis.onvoiceschanged = () => window.speechSynthesis.getVoices();
}

function getBestVoice(agentId: string): SpeechSynthesisVoice | null {
  if (!window.speechSynthesis) return null
  const vs = window.speechSynthesis.getVoices();
  if (!vs.length) return null
  const profile = AGENT_VOICE[agentId];
  const esVoices = vs.filter((v) => v.lang.startsWith("es"));
  if (!esVoices.length) return vs[0];
  if (profile?.gender === "female") {
    const fem = esVoices.find((v) =>
      v.name.toLowerCase().match(/sabina|m[oÃ³]nica|paulina|elena|laura|maria|carmen|female|mujer|femenin/)
    );
    return fem ?? esVoices[1] ?? esVoices[0];
  }
  return esVoices.find((v) => v.lang === "es-MX") ?? esVoices.find((v) => v.lang === "es-ES") ?? esVoices[0];
}

// priority=true â†’ barge-in: cancels any current speech before speaking
function speakAgent(text: string, agentId: string, priority = false): void {
  if (!window.speechSynthesis) return;
  if (priority || window.speechSynthesis.speaking) window.speechSynthesis.cancel();
  const profile = AGENT_VOICE[agentId] ?? AGENT_VOICE.SANTI;
  const u = new SpeechSynthesisUtterance(text);
  u.lang = "es-MX";
  u.pitch = profile.pitch;
  u.rate  = profile.rate;
  u.volume = 0.92;
  const voice = getBestVoice(agentId);
  if (voice) u.voice = voice;
  // Small delay on Safari/Chrome to avoid synthesis queue glitch
  setTimeout(() => window.speechSynthesis.speak(u), 50);
}

function stopSpeaking(): void {
  if (window.speechSynthesis?.speaking) window.speechSynthesis.cancel();
}
