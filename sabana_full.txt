interface ZonaConfort {

// ── 2. FUNCIONES MATEMÁTICAS Y GEOMÉTRICAS (TOPOGRAFÍA) ──────────────────────
function haversineM(lat1: number, lon1: number, lat2: number, lon2: number): number {
  const R = 6371000;
  const dLat = (lat2 - lat1) * Math.PI / 180;
  const dLon = (lon2 - lon1) * Math.PI / 180;
  const a = Math.sin(dLat/2)**2 +
            Math.cos(lat1*Math.PI/180) * Math.cos(lat2*Math.PI/180) * Math.sin(dLon/2)**2;
  return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
}

function bearing(lat1: number, lon1: number, lat2: number, lon2: number): number {
  const dLon = (lon2 - lon1) * Math.PI / 180;
  const y = Math.sin(dLon) * Math.cos(lat2 * Math.PI / 180);
  const x = Math.cos(lat1 * Math.PI / 180) * Math.sin(lat2 * Math.PI / 180) -
            Math.sin(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * Math.cos(dLon);
  return ((Math.atan2(y, x) * 180 / Math.PI) + 360) % 360;
}

function offsetPoint(lat: number, lon: number, bearingDeg: number, distM: number): [number, number] {
  const R = 6371000;
  const d = distM / R;
  const brRad = bearingDeg * Math.PI / 180;
  const lat1 = lat * Math.PI / 180;
  const lon1 = lon * Math.PI / 180;
  const lat2 = Math.asin(Math.sin(lat1)*Math.cos(d) + Math.cos(lat1)*Math.sin(d)*Math.cos(brRad));
  const lon2 = lon1 + Math.atan2(Math.sin(brRad)*Math.sin(d)*Math.cos(lat1), Math.cos(d)-Math.sin(lat1)*Math.sin(lat2));
  return [lat2 * 180 / Math.PI, lon2 * 180 / Math.PI];
}

// ── 3. ALGORITMO DE ZONAS DE CONFORT (DOMICILIO / TRABAJO) ──────────────────
// Agrupa coordenadas en una cuadrícula de ~50 metros. Las frecuencias
// nocturnas (22:00-07:59) infieren el domicilio. Las diurnas el trabajo.
function calcZonasConfort(rows: SabanaRow[]): ZonaConfort[] {
  if (!rows.length) return [];

  const GRID = 0.00045; // ~50m
  const snap = (v: number) => Math.round(v / GRID) * GRID;

  const cellMap = new Map<string, { lat: number; lon: number; night: number; day: number }>();
  
  for (const r of rows) {
    if (!r.lat || !r.lon) continue;
    const key = `${snap(r.lat).toFixed(5)},${snap(r.lon).toFixed(5)}`;
    const hour = parseInt((r.time ?? "00:00").split(":")[0], 10);
    const isNight = hour >= 22 || hour < 8;
    const prev = cellMap.get(key) ?? { lat: r.lat, lon: r.lon, night: 0, day: 0 };
    prev.night += isNight ? 1 : 0;
    prev.day   += isNight ? 0 : 1;
    cellMap.set(key, prev);
  }

  const cells = Array.from(cellMap.values());
  const topNight = cells.sort((a,b) => b.night - a.night)[0];
  const topDay   = cells.sort((a,b) => b.day   - a.day  )[0];

  const result: ZonaConfort[] = [];
  if (topNight && topNight.night > 0) {
    result.push({
      label: "DOMICILIO PROBABLE",
      lat: topNight.lat, lon: topNight.lon,
      frequency: topNight.night,
      hours: "22:00 – 07:59",
      color: "#00ffcc",
    });
  }
  if (topDay && topDay.day > 0) {
    result.push({
      label: "TRABAJO PROBABLE",
      lat: topDay.lat, lon: topDay.lon,
      frequency: topDay.day,
      hours: "08:00 – 17:59",
      color: "#c8a84b",
    });
  }
  return result;
}

// ── 4. CONSTRUCTOR DE GRAFO (ESTILO IBM i2) ─────────────────────────────────
// Cruza números de teléfono con IMEIs y alias para crear la red técnica.
function buildGraph(rows: SabanaRow[]): { nodes: GraphNode[]; links: GraphLink[] } {
  const phones = new Map<string, { count: number; alias?: string }>();
  const imeis  = new Map<string, number>();
  const edges  = new Map<string, number>();

  for (const r of rows) {
    const key = r.phone;
    const prev = phones.get(key) ?? { count: 0, alias: r.alias };
    prev.count++;
    phones.set(key, prev);

    if (r.imei) {
      imeis.set(r.imei, (imeis.get(r.imei) ?? 0) + 1);
    }

    if (r.imei && r.phone) {
      const ek = `${r.phone}||${r.imei}`;
      edges.set(ek, (edges.get(ek) ?? 0) + 1);
    }
  }

  const nodes: GraphNode[] = [];
  const links: GraphLink[] = [];

  phones.forEach(({ count, alias }, phone) => {
    nodes.push({ id: phone, label: alias ?? phone, color: "#00e5ff", group: "phone", val: Math.log2(count + 2) * 4 });
  });
  imeis.forEach((count, imei) => {
    nodes.push({ id: imei, label: `IMEI:${imei.slice(-6)}`, color: "#c8a84b", group: "imei", val: Math.log2(count + 2) * 3 });
  });
  edges.forEach((count, key) => {
    const [src, tgt] = key.split("||");
    links.push({ source: src, target: tgt, value: count, color: "#ff6d0060" });
  });

  return { nodes, links };
}

async function parseXlsx(file: File): Promise<SabanaRow[]> {
  let XLSX: any;
  try {
    XLSX = await import("xlsx");
  } catch {
    throw new Error("Librería 'xlsx' no instalada. Ejecuta: npm install xlsx");
  }
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = (e) => {
      try {
        const data = new Uint8Array(e.target!.result as ArrayBuffer);
        const wb   = XLSX.read(data, { type: "array" });
        const ws   = wb.Sheets[wb.SheetNames[0]];
        const json: any[] = XLSX.utils.sheet_to_json(ws, { defval: "" });

        const rows: SabanaRow[] = json.map((r: any) => {
          const kv = (keys: string[], fallback: string) =>
            (keys.find((k) => r[k] !== undefined) && r[keys.find((k) => r[k] !== undefined)!]) ?? fallback;
          return {
            phone:    String(kv(["telefono","Telefono","phone","Phone","numero","Numero"], "")),
            date:     String(kv(["fecha","Fecha","date","Date"], "")),
            time:     String(kv(["hora","Hora","time","Time"], "00:00")),
            duration: Number(kv(["duracion","Duracion","duration","Duration"], 0)),
            type:     String(kv(["tipo","Tipo","type","Type"], "DATOS")),
            imei:     String(kv(["imei","IMEI","Imei"], "")),
            cellId:   String(kv(["cellid","CellId","CellID","celda","Celda"], "")),
            lac:      String(kv(["lac","LAC"], "")),
            lat:      parseFloat(String(kv(["lat","Lat","latitud","Latitud","latitude"], "0"))) || 0,
            lon:      parseFloat(String(kv(["lon","Lon","lng","Lng","longitud","Longitud","longitude"], "0"))) || 0,
            azimuth:  parseFloat(String(kv(["azimuth","Azimuth","azimut","Azimut"], "0"))) || 0,
            address:  String(kv(["direccion","Direccion","address","Address"], "")),
            alias:    String(kv(["alias","Alias"], "")),
          };
        }).filter((r) => r.phone);

        resolve(rows);
      } catch (err: any) {
        reject(new Error(`Error parseando Excel: ${err.message}`));
      }
    };
    reader.onerror = () => reject(new Error("Error leyendo archivo"));
    reader.readAsArrayBuffer(file);
  });
}

// ── 6. GRÁFICO DE RESPALDO (SVG) Y CARGA DINÁMICA DE FORCE GRAPH ────────────
const FallbackGraph: React.FC<{ nodes: GraphNode[]; links: GraphLink[] }> = ({ nodes, links }) => {
  if (!nodes.length) return <p style={{ color: "#3a6a5a", fontSize: 9, padding: 12 }}>Sin datos de grafo. Sube una sábana.</p>;
  const W = 540, H = 340;
  const positioned = nodes.map((n, i) => {
    const angle = (2 * Math.PI * i) / nodes.length;
    const r = Math.min(W, H) / 2 - 32;
    return { ...n, cx: W/2 + r * Math.cos(angle), cy: H/2 + r * Math.sin(angle) };
  });
  const posMap = new Map(positioned.map((n) => [n.id, n]));

  return (
    <svg width="100%" viewBox={`0 0 ${W} ${H}`} style={{ background: "rgba(0,4,12,0.98)", borderRadius: 8 }}>
      {links.map((l, i) => {
        const s = posMap.get(String(l.source));
        const t = posMap.get(String(l.target));
        if (!s || !t) return null
        return <line key={i} x1={s.cx} y1={s.cy} x2={t.cx} y2={t.cy} stroke={l.color} strokeWidth={Math.log2(l.value+1)} />;
      })}
      {positioned.map((n) => (
        <g key={n.id}>
          <circle cx={n.cx} cy={n.cy} r={n.val + 4} fill={`${n.color}22`} stroke={n.color} strokeWidth={1.5} />
          <text x={n.cx} y={n.cy + 1} textAnchor="middle" dominantBaseline="middle"
            fontSize={8} fontFamily="monospace" fill={n.color} fontWeight="700">
            {n.label.slice(0, 10)}
          </text>
        </g>
      ))}
    </svg>
  );
};
